// Interpreter.cpp:456
Tensor* Interpreter::getSessionOutput(const Session* session, const char* name) {
    if (session == nullptr) return nullptr;
    std::unique_lock<std::mutex> _l(mNet->lock);       // ① 与 createSession/resizeSession 互斥
    auto tensor = session->getOutput(name);              // ② 查 Session 内部 map
    mNet->tensorMap.insert({tensor, session});           // ③ 记录 Tensor→Session 归属
    return tensor;                                       // ④ 返回 Device Tensor
}

// Session.cpp:426
Tensor* Session::getOutput(const char* name) const {
    MNN_ASSERT(!mInfo.outputTensor.empty());
    if (nullptr == name) {
        return mInfo.outputTensor.begin()->second;       // 无名字 → 第一个输出
    }
    auto iter = mInfo.outputTensor.find(name);           // 按 tensorName 查找
    if (iter == mInfo.outputTensor.end()) {
        MNN_PRINT("Error: can't find output: %s\n", name);
        return nullptr;
    }
    return iter->second;
}

读取数据：copyToHostTensor 深度分析

方式一：copyToHostTensor（推荐，跨后端通用）

// 步骤:

//    复制 shape/dim → setLinearLayout → malloc(size) → host 有值

// ===== copyToHostTensor 内部实现 (Tensor.cpp:187) =====
bool Tensor::copyToHostTensor(Tensor* hostTensor) const {
    auto bn = mDescribe->getBackend();   // 获取后端
    if (nullptr == bn) return false;
    bn->onCopyBuffer(this, hostTensor);  // 委托给 Backend
    return true;
}

方式二：直接读 host 指针（仅 CPU 后端）

float* data = output->host<float>();
// ⚠️ 仅在 CPU 后端保证 host 有值
// ⚠️ GPU 后端 host 为 nullptr，需用 copyToHostTensor
// ⚠️ 数据是 NC4HW4 格式！直接读会得到错误排列

方式三：Tensor::map / unmap（高级用法，零拷贝）

// map: 获取 GPU Tensor 的 host 可读指针
void* ptr = tensor->map(Tensor::MAP_TENSOR_READ, Tensor::CAFFE);
float* data = (float*)ptr;  // 直接读

// unmap: 释放映射（WRITE 模式会写回 GPU）
tensor->unmap(Tensor::MAP_TENSOR_READ, Tensor::CAFFE, ptr);

// 内部流程:
// MAP_TENSOR_READ:
//   ① Backend::onSync(READ, false, this) — 确保 GPU 计算完成
//   ② Backend::onMapTensor — 尝试 GPU 直接映射
//   ③ 若失败: malloc → onCopyBuffer(device→host)
// MAP_TENSOR_WRITE:
//   ① onMapTensor / malloc
//   ② unmap 时 onCopyBuffer(host→device) 写回

CPU Backend::onCopyBuffer — 完整数据搬运流程

// CPUBackend.cpp:778
void CPUBackend::onCopyBuffer(const Tensor* src, const Tensor* dst) const {
    _resetDynamicMemory();

    // Step 1: 维度检查
    //   src 和 dst 维度数不同 → 仅当最后一维 extent=1 才允许
    //   src 和 dst 维度类型相同 → 断言各维 extent 匹配

    // Step 2: host 指针检查
    if (nullptr == srcBuffer.host || nullptr == dstBuffer.host) return;

    // Step 3: 数据类型转换（如果 src/dst 类型不同）
    if (getDataType(src) != getDataType(dst)) {
        // 创建临时 Tensor，用 CPUCastCreator::cast 做类型转换
        //   FLOAT→INT8 或 INT8→FLOAT
        wrapTensor.reset(Tensor::createDevice(src->shape(), dst->getType()));
        wrapTensor->buffer().host = malloc(...);
        CPUCastCreator::cast(src, wrapTensor, this, convertType);
        src = wrapTensor.get();  // 用转换后的作为源
    }

    // Step 4: 数据格式转换（核心）
    auto code = CPUTensorConverter::convert(src, dst);
    //   NC4HW4 → NCHW / NHWC 或其他格式组合
}

// CPU 端 onCopyBuffer 做了两件事:
//   1. 类型转换（如 INT8→FLOAT）
//   2. 数据布局转换（NC4HW4 → NCHW/NHWC）

// ===== copyFromHostTensor 内部实现 (Tensor.cpp:171) =====
bool Tensor::copyFromHostTensor(const Tensor* hostTensor) {
    auto bn = mDescribe->getBackend();
    if (nullptr == bn) return false;

    // 跨后端处理: hostTensor 也有自己的 Backend?
    auto hostbn = hostTensor->mDescribe->getBackend();
    if (nullptr != hostbn && hostbn->type() != bn->type()
                         && hostbn->type() != MNN_FORWARD_CPU) {
        // 先拷贝到临时 CPU Host Tensor
        tmpTensor.reset(new Tensor(hostTensor, hostTensor->getDimensionType()));
        hostTensor->copyToHostTensor(tmpTensor.get());
        hostTensor = tmpTensor.get();
    }

    bn->onCopyBuffer(hostTensor, this);  // 委托 Backend
    return true;
}

面试重点

Host Tensor 和 Device Tensor 的本质区别是什么？
Host Tensor：mDescribe->memoryType = MEMORY_HOST，mBuffer.host 指向 malloc 内存，用户可任意读写，析构时自动 free。Device Tensor：memoryType = MEMORY_BACKEND，mBuffer.host 可能为空，实际数据在 GPU/后端上，需通过 copyToHostTensor 或 map 才能读到 CPU。Session 内所有 Tensor 都是 Device Tensor。
copyToHostTensor 内部做了什么？
① 获取 Tensor 关联的 Backend ② 调用 Backend::onCopyBuffer(this, hostTensor)。CPU 端做两件事：类型转换（如 INT8→FLOAT）+ 数据布局转换（NC4HW4→NCHW/NHWC）。GPU 端先确保 GPU 计算完成，再通过 DMA 拷贝到 host。
为什么不直接读 output->host()？
两个原因：① GPU 后端 host 为 nullptr（数据在显存上）② CPU 后端虽然有 host，但数据是 NC4HW4 格式（通道按 4 打包），直接当 NCHW 读会得到错误排列。必须经过 copyToHostTensor 做格式转换。
getSessionOutput 返回的 Tensor 和 Session 的关系？
返回的是 Session::mInfo.outputTensor 中的指针，生命周期与 Session 绑定。Session 释放后指针失效。Interpreter 通过 tensorMap 记录此关系，用于 waitSessionFinish 和 releaseSession。
NC4HW4 是什么，为什么用它？
NC4HW4 = 通道维度按 4 打包到宽度维度之前。4 通道一组对齐到 SIMD 128-bit 寄存器，消除跨通道 stride 开销。所有引擎内部数据都用 NC4HW4，输入时 NCHW→NC4HW4（Pack），输出时 NC4HW4→NCHW（Unpack），内部无转换。
cross-backend copy 怎么处理？
copyFromHostTensor 中检查源 Tensor 的 Backend 类型，如果是非 CPU 非目标类型的后端，先 copyToHostTensor 到临时 CPU Host Tensor（走 GPU→CPU），再 copyFromHostTensor（走 CPU→目标后端）。Copy 以 CPU 为中转站。
Tensor::map 和 copyToHostTensor 的区别？
map：零拷贝映射，直接拿到 GPU 内存的 host 可访问指针（通过 OpenCL map/unmap 或 Metal 的 contents），适合一次性读取。copyToHostTensor：完整拷贝一份到新内存，数据转为用户格式（NCHW），适合持久保存或多次访问。map/unmap 必须配对使用。
5 步推理循环中哪些调用需要锁？
getSessionInput/getSessionOutput：mutex lock（防止与 createSession 并发修改 tensorMap）。createSession/resizeSession/runSession/releaseSession：mutex lock（同一 Interpreter 串行）。仅 runSessionWithCallBack 默认不加锁（由调用者保证安全）。

深度总结

✅ Tensor = halide_buffer_t (公开) + InsideDescribe (私有元数据)，两层结构分离数据与语义
✅ MemoryType 4 种: BACKEND (Backend管理) / HOST (malloc) / VIRTUAL (无内存) / OUTSIDE (外部指针)
✅ Usage 5 种: NORMAL / INPUT / OUTPUT / CONSTANT / TRAINABLE，标记 Tensor 语义角色
✅ getSessionOutput: mutex → Session::getOutput(name) → outputTensor map 查找 → tensorMap 记录归属
✅ copyToHostTensor: getBackend → onCopyBuffer → 类型转换(Cast) + 格式转换(NC4HW4→NCHW)
✅ copyFromHostTensor: getBackend → 跨后端中转(如GPU→CPU→目标) → onCopyBuffer
✅ CPUBackend::onCopyBuffer 核心: 类型 Cast (FLOAT/INT8) + CPUTensorConverter convert (NC4HW4↔NCHW↔NHWC)
✅ map/unmap 零拷贝路径: onSync 确保完成 → onMapTensor 尝试映射 → 失败则 malloc 中转
✅ tensorMap: Interpreter 维护 Tensor→Session 映射，用于 waitSessionFinish 等待 GPU 操作
✅ 5 步循环: load → createSession → setInput+resize → runSession → getOutput+copyToHostTensor
✅ 数据格式转换仅在边界发生: 输入 NCHW→NC4HW4, 输出 NC4HW4→NCHW, 内部全程 NC4HW4
